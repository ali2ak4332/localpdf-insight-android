cmake_minimum_required(VERSION 3.22.1)
project(apt_reader LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(NOT APT_LLAMA_SOURCE)
  include(FetchContent)
  FetchContent_Declare(apt_llama
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/df9beac7153166550571c70cc667f270b745da49
    URL_HASH SHA256=35c3811bef57f5e5e3d09cce9b487ca6adeeb6783cdb259712075ee36f38b4e8)
  FetchContent_GetProperties(apt_llama)
  if(NOT apt_llama_POPULATED)
    FetchContent_Populate(apt_llama)
  endif()
  set(APT_LLAMA_SOURCE ${apt_llama_SOURCE_DIR})
endif()
add_subdirectory(${APT_LLAMA_SOURCE} ${CMAKE_CURRENT_BINARY_DIR}/llama EXCLUDE_FROM_ALL)
add_library(apt_engine STATIC apt_engine.cpp)
target_include_directories(apt_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(apt_engine PUBLIC llama)
if(ANDROID)
  add_library(apt_reader SHARED apt_jni.cpp)
  target_link_libraries(apt_reader PRIVATE apt_engine android log)
  target_link_options(apt_reader PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(apt-select ../../../../../tests/apt/native_main.cpp)
  target_link_libraries(apt-select PRIVATE apt_engine)
  add_executable(apt-cancel-test ../../../../../tests/apt/native_cancel.cpp)
  target_link_libraries(apt-cancel-test PRIVATE apt_engine)
endif()
